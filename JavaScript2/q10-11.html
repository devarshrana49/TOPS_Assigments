// Primitive type let user1 = "Priya"; let user2 = user1; user1 = "Riya"; console.log(user1); console.log(user2); // Reference type let favApps1 = ["Instagram", "Spotify"]; let favApps2 = favApps1; favApps1.push("Zomato"); console.log(favApps1); console.log(favApps2); // Strings are primitive, so user2 keeps the old value. // Arrays are reference types, so both variables refer to the same array.
